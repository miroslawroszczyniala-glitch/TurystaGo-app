// ============================================ // TURYSTAGO // PLAN MAP – RZECZYWISTY PLAN UŻYTKOWNIKA // ============================================ let touristMap = null; let touristRoute = null; let touristMarkers = []; // ============================================ // ODCZYT PLANU Z moj-plan.html // ============================================ function getSavedPlan() { const saved = localStorage.getItem("turystagoPlan"); if (!saved) { return null; } try { return JSON.parse(saved); } catch (error) { console.error( "Błąd odczytu planu:", error ); return null; } } // ============================================ // OTWARCIE MAPY // ============================================ function openPlanMap() { const planScreen = document.getElementById("plan-screen"); const mapScreen = document.getElementById("map-screen"); if (planScreen) { planScreen.style.display = "none"; } if (mapScreen) { mapScreen.style.display = "block"; } createTouristMap(); } // ============================================ // UTWORZENIE MAPY // ============================================ function createTouristMap() { if (typeof L === "undefined") { alert( "Nie udało się załadować mapy." ); return; } const savedPlan = getSavedPlan(); if (!savedPlan) { alert( "Nie znaleziono przygotowanego planu." ); return; } if ( !savedPlan.places || savedPlan.places.length === 0 ) { alert( "Twój plan nie zawiera jeszcze atrakcji." ); return; } if (touristMap) { touristMap.remove(); touristMap = null; } touristMap = L.map("tourist-map"); L.tileLayer( "https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png", { maxZoom: 19, attribution: "© OpenStreetMap" } ).addTo(touristMap); createRoute( savedPlan.places ); } // ============================================ // TWORZENIE TRASY // ============================================ function createRoute(places) { if ( !touristMap || !places || places.length === 0 ) { return; } touristMarkers.forEach( function(marker) { touristMap.removeLayer(marker); } ); touristMarkers = []; if (touristRoute) { touristMap.removeLayer( touristRoute ); touristRoute = null; } const coordinates = []; places.forEach( function(place, index) { /* * Jeżeli atrakcja ma współrzędne, * używamy ich. * * Jeżeli jeszcze ich nie ma, * nie pokazujemy jej jako fałszywego * punktu na mapie. */ if ( typeof place.lat !== "number" || typeof place.lng !== "number" ) { return; } const position = [ place.lat, place.lng ]; coordinates.push(position); const marker = L.marker(position) .addTo(touristMap) .bindPopup( "" + (index + 1) + ". " + place.name + "" ); touristMarkers.push( marker ); } ); if (coordinates.length === 0) { showRouteInformation( places ); return; } /* * Na tym etapie pokazujemy trasę * pomiędzy punktami. * * Następnie podłączymy prawdziwe * wyznaczanie trasy drogowej. */ touristRoute = L.polyline( coordinates, { weight: 6, opacity: 0.85 } ).addTo(touristMap); touristMap.fitBounds( touristRoute.getBounds(), { padding: [30, 30] } ); showRouteInformation( places ); } // ============================================ // INFORMACJE O TRASIE // ============================================ function showRouteInformation( places ) { const routeInfo = document.getElementById( "route-information" ); if (!routeInfo) { return; } const savedPlan = getSavedPlan(); let html = `

🗺️ Twój plan zwiedzania

`; if (savedPlan) { if (savedPlan.start) { html += `

📍 Start: ${savedPlan.start}

`; } if (savedPlan.time) { html += `

⏱️ Czas: ${savedPlan.time}

`; } } html += `
    `; places.forEach( function(place) { html += `
  1. ${place.name} `; if (place.time) { html += `
    ⏱️ ${place.time} `; } if (place.distance) { html += `
    📍 ${place.distance} `; } html += `
    2. `; } ); html += `
`; routeInfo.innerHTML = html; } // ============================================ // ROZPOCZĘCIE ZWIEDZANIA // ============================================ function startTour() { alert( "TurystaGo rozpoczyna zwiedzanie.\n\n" + "Następny etap: prowadzenie GPS " + "od punktu do punktu." ); } // ============================================ // POWRÓT DO PLANU // ============================================ function backToPlan() { const mapScreen = document.getElementById( "map-screen" ); if (mapScreen) { mapScreen.style.display = "none"; } window.location.href = "moj-plan.html"; }
